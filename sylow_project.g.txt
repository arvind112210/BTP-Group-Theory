L := [];
ClassIndex := function(H, classes)
    local i; 
    for i in [1..Length(classes)] do 
        if IsConjugate(G, Representative(classes[i]), H) then 
            return i; 
        fi; 
    od; 
    return -1;
end;

for i in [1..Length(cc)] do
    L[i] := rec( id:=i, order:=Size(Representative(cc[i])), 
                 classSize:=Size(cc[i]), 
                 struct:=StructureDescription(Representative(cc[i])), 
                 parents:=[] );
od;

for i in [1..Length(cc)] do
    for M in MaximalSubgroups(Representative(cc[i])) do
        idxM := ClassIndex(M, cc);
        if idxM <> -1 and not (i in L[idxM].parents) then 
            Add(L[idxM].parents, i); 
        fi;
    od;
od;

GenerateConjugationMasterclass(L, G, "S4_BTP_Lattice_Final.svg");
