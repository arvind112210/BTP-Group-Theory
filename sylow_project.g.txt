# 1. DEFINE RENDERER
GenerateConjugationMasterclass := function(L, G, fn)
    local out, h, ranks, ord, node, r, idx, x, y, 
          px, py, cx, cy, pId, tR, tX, 
          cL2, cL3, sSub2, sSub3, nSyl2, nSyl3,
          conjStrs2, conjStrs3, i, j, g, genList, 
          genStr, testElems, elemStrs, 
          map_g, mappings2, mappings3, conjP,
          N2, szN2, N3, szN3, genListN, dispName, GetX, GetY;

    sSub2 := SylowSubgroup(G, 2); 
    cL2 := ConjugateSubgroups(G, sSub2); 
    nSyl2 := Length(cL2);
    conjStrs2 := [];
    for i in [1..nSyl2] do
        genList := GeneratorsOfGroup(cL2[i]); 
        genStr := "";
        for g in genList do 
            genStr:=Concatenation(genStr,String(g),", "); 
        od;
        if Length(genStr)>1 then 
            genStr:=genStr{[1..Length(genStr)-2]}; 
        fi;
        Add(conjStrs2, Concatenation(
            "P", String(i), " = &#9001; ", genStr, " &#9002;"
        ));
    od;

    sSub3 := SylowSubgroup(G, 3); 
    cL3 := ConjugateSubgroups(G, sSub3); 
    nSyl3 := Length(cL3);
    conjStrs3 := [];
    for i in [1..nSyl3] do
        genList := GeneratorsOfGroup(cL3[i]); 
        genStr := "";
        for g in genList do 
            genStr:=Concatenation(genStr,String(g),", "); 
        od;
        if Length(genStr)>1 then 
            genStr:=genStr{[1..Length(genStr)-2]}; 
        fi;
        Add(conjStrs3, Concatenation(
            "Q", String(i), " = &#9001; ", genStr, " &#9002;"
        ));
    od;
    
    testElems := [ (1,2,3,4), (1,2,3), (1,2) ];
    elemStrs := [ "(1,2,3,4)", "(1,2,3)", "(1,2)" ];
    
    mappings2 := [];
    for g in testElems do
        map_g := [];
        for i in [1..nSyl2] do
            conjP := cL2[i]^(g^-1);
            for j in [1..nSyl2] do 
                if conjP = cL2[j] then Add(map_g, j); break; fi; 
            od;
        od; 
        Add(mappings2, map_g);
    od;

    mappings3 := [];
    for g in testElems do
        map_g := [];
        for i in [1..nSyl3] do
            conjP := cL3[i]^(g^-1);
            for j in [1..nSyl3] do 
                if conjP = cL3[j] then Add(map_g, j); break; fi; 
            od;
        od; 
        Add(mappings3, map_g);
    od;

    N2 := Normalizer(G, cL2[1]); szN2 := Size(N2);
    genListN := GeneratorsOfGroup(N2); genStr := "";
    for g in genListN do 
        genStr:=Concatenation(genStr,String(g),", "); 
    od;

        genStr:=genStr{[1..Length(genStr)-2]}; 
    fi;

    genListN := GeneratorsOfGroup(N3); genStr := "";

        genStr:=Concatenation(genStr,String(g),", "); 
    od;

        genStr:=genStr{[1..Length(genStr)-2]}; 
    fi;

G := SymmetricGroup(4);
cc := ConjugacyClassesSubgroups(G);
L := [];
ClassIndex := function(H, classes)
    local i; 
    for i in [1..Length(classes)] do 
        if IsConjugate(G, Representative(classes[i]), H) then 
            return i; 
        fi; 
    od; 
    return -1;
end;

for i in [1..Length(cc)] do
    L[i] := rec( id:=i, order:=Size(Representative(cc[i])), 
                 classSize:=Size(cc[i]), 
                 struct:=StructureDescription(Representative(cc[i])), 
                 parents:=[] );
od;

for i in [1..Length(cc)] do
    for M in MaximalSubgroups(Representative(cc[i])) do
        idxM := ClassIndex(M, cc);
        if idxM <> -1 and not (i in L[idxM].parents) then 
            Add(L[idxM].parents, i); 
        fi;
    od;
od;

GenerateConjugationMasterclass(L, G, "S4_BTP_Lattice_Final.svg");
